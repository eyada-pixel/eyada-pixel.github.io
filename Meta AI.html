    import React, { useState, useEffect } from 'react';

export default function TodaySessions() {
  const [sessions, setSessions] = useState([]);

  useEffect(() => {
    // هسه حاليا بيانات وهمية - بعدين نربطها بقاعدة البيانات مالتك
    const allSessions = JSON.parse(localStorage.getItem('sessions') || '[]');

    const today = new Date().toISOString().split('T')[0]; // تاريخ اليوم

    const todayOnly = allSessions.filter(s => s.date === today);
    setSessions(todayOnly);
  }, []);

  return (
    <div style={{ padding: '20px', fontFamily: 'sans-serif', direction: 'rtl' }}>
      <h1 style={{ textAlign: 'center' }}>🗓️ جلسات اليوم - {new Date().toLocaleDateString('ar-EG')}</h1>

      {sessions.length === 0? (
        <p style={{ textAlign: 'center', marginTop: '30px', color: '#888' }}>لا توجد جلسات اليوم</p>
      ) : (
        <table style={{ width: '100%', borderCollapse: 'collapse', marginTop: '20px' }}>
          <thead>
            <tr style={{ background: '#0d6efd', color: 'white' }}>
              <th style={{ padding: '12px', border: '1px solid #ddd' }}>الوقت</th>
              <th style={{ padding: '12px', border: '1px solid #ddd' }}>اسم المريض</th>
              <th style={{ padding: '12px', border: '1px solid #ddd' }}>نوع العلاج</th>
              <th style={{ padding: '12px', border: '1px solid #ddd' }}>الحالة</th>
            </tr>
          </thead>
          <tbody>
            {sessions.map((s, i) => (
              <tr key={i} style={{ background: i % 2 === 0? '#f2f2f2' : 'white' }}>
                <td style={{ padding: '10px', border: '1px solid #ddd', textAlign: 'center' }}>{s.time}</td>
                <td style={{ padding: '10px', border: '1px solid #ddd', textAlign: 'center' }}>{s.name}</td>
                <td style={{ padding: '10px', border: '1px solid #ddd', textAlign: 'center' }}>{s.type}</td>
                <td style={{ padding: '10px', border: '1px solid #ddd', textAlign: 'center' }}>{s.status}</td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </div>
  );
}
